🎯Numéroter les commandes, client par client
requete.sql
SELECT clients.prenom, clients.nom, produits.nom AS produit, commandes.date_commande, ROW_NUMBER() OVER ( ORDER BY commandes.date_commande ) AS numeroFROM commandesJOIN clients ON clients.id = commandes.client_idJOIN produits ON produits.id = commandes.produit_id;
